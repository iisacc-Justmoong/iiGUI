cmake_minimum_required(VERSION 3.31)
project(HelloLVUI LANGUAGES CXX)
include(CTest)
if(NOT TARGET iiGUI::iiGUI)
    find_package(iiGUI 0.4 CONFIG REQUIRED)
endif()
if(NOT TARGET LVUI::LVUI)
    find_package(LVUI 0.1 CONFIG REQUIRED)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS main.lvui)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/main.lvui" LVUI_SOURCE)
configure_file(ViewSource.h.in "${CMAKE_CURRENT_BINARY_DIR}/ViewSource.h" @ONLY)
add_executable(HelloLVUI main.cpp)
target_link_libraries(HelloLVUI PRIVATE iiGUI::iiGUI LVUI::LVUI)
target_include_directories(HelloLVUI PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(BUILD_TESTING)
    add_test(NAME HelloLVUI.native COMMAND HelloLVUI --self-test)
    set_tests_properties(HelloLVUI.native PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
    add_test(NAME HelloLVUI.vulkan COMMAND HelloLVUI --self-test --vulkan)
    set_tests_properties(HelloLVUI.vulkan PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
endif()
