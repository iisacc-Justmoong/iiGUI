cmake_minimum_required(VERSION 3.31)
project(HelloAndroid LANGUAGES C CXX)
option(IIGUI_ANDROID_OS_SMOKE "Verify clipboard preservation and Choreographer on a physical device" OFF)
option(IIGUI_ANDROID_RENDER_SMOKE "Verify retained label pixels across Vulkan updates" OFF)
if(NOT ANDROID)
    message(FATAL_ERROR "HelloAndroid requires the Android NDK toolchain")
endif()
if(NOT TARGET iiGUI::iiGUI)
    find_package(iiGUI 0.4 CONFIG REQUIRED)
endif()
if(NOT TARGET LVUI::Native)
    find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME HelloWorld SOURCE HelloWorld.lvui OUTPUT_VARIABLE _view)
add_library(HelloAndroid SHARED main.cpp android_main.cpp "${_view}"
    "${CMAKE_ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
target_include_directories(HelloAndroid PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${CMAKE_ANDROID_NDK}/sources/android/native_app_glue")
target_link_libraries(HelloAndroid PRIVATE iiGUI::iiGUI LVUI::Native android log)
if(IIGUI_ANDROID_RENDER_SMOKE)
    target_sources(HelloAndroid PRIVATE ../../tests/android_retained_labels.cpp)
    target_compile_definitions(HelloAndroid PRIVATE IIGUI_ANDROID_RENDER_SMOKE=1)
endif()
if(IIGUI_ANDROID_OS_SMOKE)
    target_sources(HelloAndroid PRIVATE ../../tests/android_os.cpp)
    target_include_directories(HelloAndroid PRIVATE ../../src)
    target_compile_definitions(HelloAndroid PRIVATE IIGUI_ANDROID_OS_SMOKE=1)
endif()
target_compile_features(HelloAndroid PRIVATE cxx_std_23)
target_link_options(HelloAndroid PRIVATE "LINKER:-u,ANativeActivity_onCreate")
set_target_properties(HelloAndroid PROPERTIES OUTPUT_NAME hello_lvui CXX_EXTENSIONS OFF)
