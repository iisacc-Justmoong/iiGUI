cmake_minimum_required(VERSION 3.31)
project(HelloHosts LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_SCAN_FOR_MODULES)
    # This sample uses ordinary C++23 translation units. Skip unused module
    # scanning in native and Emscripten kits; preserve an explicit app setting.
    set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
endif()
if(NOT TARGET iiGUI::iiGUI)
    find_package(iiGUI 0.4 CONFIG REQUIRED)
endif()
if(NOT TARGET LVUI::Native)
    find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
endif()
lvui_generate(NAME HelloHosts SOURCE HelloHosts.lvui OUTPUT_VARIABLE _view INCLUDES HostActions.h)
iisdkcore_add_application(HelloHosts GUI SOURCES main.cpp "${_view}" LIBRARIES iiGUI::iiGUI LVUI::Native RESOURCES ${iiGUI_NATIVE_RESOURCES})
target_include_directories(HelloHosts PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
if(EMSCRIPTEN)
    set_target_properties(HelloHosts PROPERTIES SUFFIX ".html")
    target_link_options(HelloHosts PRIVATE "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/shell.html")
    foreach(_asset IN LISTS iiGUI_NATIVE_RESOURCES)
        get_filename_component(_name "${_asset}" NAME)
        target_link_options(HelloHosts PRIVATE "SHELL:--preload-file ${_asset}@/resources/${_name}")
    endforeach()
    target_link_options(HelloHosts PRIVATE -sSTACK_SIZE=1048576 -sINITIAL_MEMORY=134217728)
endif()
install(TARGETS HelloHosts BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(EMSCRIPTEN)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/HelloHosts.js" "${CMAKE_CURRENT_BINARY_DIR}/HelloHosts.wasm" "${CMAKE_CURRENT_BINARY_DIR}/HelloHosts.data" DESTINATION bin)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/HelloHosts.worker.js" DESTINATION bin OPTIONAL)
elseif(NOT APPLE)
    install(FILES ${iiGUI_NATIVE_RESOURCES} DESTINATION bin/resources)
endif()
