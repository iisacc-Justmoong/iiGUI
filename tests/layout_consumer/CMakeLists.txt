cmake_minimum_required(VERSION 3.31)
project(iiGUILayoutConsumer LANGUAGES CXX)
find_package(iiGUI 0.4 CONFIG REQUIRED)
find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
lvui_generate(NAME LayoutPackage SOURCE Layout.lvui INCLUDES Model.h OUTPUT_VARIABLE _view)
lvui_generate(NAME WindowPackage SOURCE Window.lvui INCLUDES Model.h OUTPUT_VARIABLE _window)
lvui_generate(NAME ReadonlyPackage SOURCE Readonly.lvui INCLUDES Model.h OUTPUT_VARIABLE _readonly)
lvui_generate(NAME ObjectInputPackage SOURCE ObjectInput.lvui INCLUDES Model.h OUTPUT_VARIABLE _object_input)
lvui_generate(NAME MethodPackage SOURCE Methods.lvui INCLUDES Model.h OUTPUT_VARIABLE _method)
lvui_generate(NAME MethodReferencePackage SOURCE MethodReferences.lvui INCLUDES Model.h OUTPUT_VARIABLE _method_reference)
add_executable(layout_consumer main.cpp "${_view}" "${_window}" "${_readonly}" "${_object_input}" "${_method}" "${_method_reference}")
target_include_directories(layout_consumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(layout_consumer PRIVATE iiGUI::iiGUI LVUI::Native)
set_target_properties(layout_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME iiGUI.LayoutPackage COMMAND layout_consumer)
