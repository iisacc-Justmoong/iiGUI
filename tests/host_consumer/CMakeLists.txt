cmake_minimum_required(VERSION 3.31)
project(iiGUIHostConsumer LANGUAGES CXX)
include(CTest)
find_package(iiGUI 0.4 CONFIG REQUIRED)
if(DEFINED EXPECT_HOST_BACKEND AND NOT iiGUI_HOST_BACKEND STREQUAL EXPECT_HOST_BACKEND)
    message(FATAL_ERROR "Installed iiGUI host is ${iiGUI_HOST_BACKEND}, expected ${EXPECT_HOST_BACKEND}")
endif()
add_executable(iiGUIHostConsumer main.cpp)
target_link_libraries(iiGUIHostConsumer PRIVATE iiGUI::iiGUI)
add_test(NAME iiGUI.host_installed_consumer COMMAND iiGUIHostConsumer)
set_tests_properties(iiGUI.host_installed_consumer PROPERTIES TIMEOUT 15)
