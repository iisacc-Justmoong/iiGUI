cmake_minimum_required(VERSION 3.31)
project(iiGUIPlatformContracts LANGUAGES CXX)
include(CTest)
find_package(iiSDKCore CONFIG REQUIRED COMPONENTS Contracts)
get_filename_component(_source "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(IIGUI_GENERATED_INCLUDE "" CACHE PATH "Generated iiGUI Export.h include root from a configured native SDK build")
option(IIGUI_CONTRACT_SANITIZE "Instrument platform contracts with ASan and UBSan" OFF)
if(NOT EXISTS "${IIGUI_GENERATED_INCLUDE}/iiGUI/Export.h")
    message(FATAL_ERROR "Configure iiGUI first and supply IIGUI_GENERATED_INCLUDE")
endif()
foreach(_case host_contract wayland_policy host_input_queue pipe_write surface_registry pending_callbacks font_face_reference)
    if(_case STREQUAL "pipe_write" AND NOT UNIX)
        continue()
    endif()
    add_executable(${_case} "${_source}/tests/${_case}.cpp")
    if(_case STREQUAL "host_contract")
        target_sources(${_case} PRIVATE "${_source}/src/iiGUI/HostCapabilities.cpp")
    endif()
    target_link_libraries(${_case} PRIVATE iiSDKCore::Contracts)
    target_include_directories(${_case} PRIVATE "${_source}/src" "${IIGUI_GENERATED_INCLUDE}")
    target_compile_definitions(${_case} PRIVATE IIGUI_STATIC_DEFINE)
    if(IIGUI_CONTRACT_SANITIZE)
        target_compile_options(${_case} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${_case} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${_case} COMMAND ${_case})
    set_tests_properties(${_case} PROPERTIES TIMEOUT 15)
endforeach()
